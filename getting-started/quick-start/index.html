    Quick Start (빠른 시작)¶

    Prism을 사용하여 5분 안에 Spring Boot 프로젝트에 A/B 테스트를 적용하는 방법을 알아봅니다.

    전제 조건 (Prerequisites)¶

    • JDK 21 이상
    • Spring Boot 3.x 이상
    • Gradle 또는 Maven
    • Docker (예제 실행 시)

    방법 1: 예제 프로젝트로 시작 (권장)¶

    가장 빠르게 Prism을 체험하는 방법입니다.

    # 전체 서비스(DB, API, Admin, Examples) 원클릭 실행
./prism-examples/scripts/quick-start.sh

    실행 후 접속 가능한 주소: - Admin: http://localhost:8070 (실험 관리) - API: http://localhost:8080 (트래픽 분배) - Examples: http://localhost:8090 (온라인 서점 예제)

    방법 2: 직접 프로젝트에 적용¶

    1단계: 의존성 추가¶

    build.gradle.kts에 Prism Starter 의존성을 추가합니다.

    dependencies {
    implementation("io.github.silbaram:prism-spring-boot-starter:0.1.0")
}

    2단계: 설정 추가¶

    application.yml에 Prism API 서버 주소를 설정합니다.

    prism:
  api:
    url: "http://localhost:8080"
    timeout: 3000
  fallback:
    enabled: true  # API 장애 시 기본값 사용

    3단계: 실험 적용 (Experiment)¶

    컨트롤러 메서드에 @PrismExperiment를 추가하여 트래픽을 분배합니다.

    import io.github.silbaram.prism.starter.annotation.PrismExperiment
import io.github.silbaram.prism.starter.annotation.PrismUserId
import io.github.silbaram.prism.starter.aop.PrismContext

@RestController
@RequestMapping("/shop")
class ProductController {

    @GetMapping("/products/{id}")
    @PrismExperiment(experimentKey = "buy_button_color")
    fun getProduct(
        @PathVariable id: Long,
        @PrismUserId @RequestParam userId: String
    ): ProductResponse {
        // 현재 할당된 변형(Variant) 가져오기
        val variant = PrismContext.getCurrentVariant() ?: "CONTROL"

        return ProductResponse(
            id = id,
            name = "상품 $id",
            buttonColor = if (variant == "RED") "red" else "blue"
        )
    }
}

    핵심 포인트: - @PrismExperiment: 실험 키를 지정하여 자동 트래픽 분배 - @PrismUserId: 사용자 식별자 자동 추출 - PrismContext.getCurrentVariant(): 할당된 변형 조회

    4단계: 전환 추적 (Conversion Tracking)¶

    사용자의 행동(클릭, 구매 등)을 추적합니다.

    import io.github.silbaram.prism.starter.annotation.PrismTrackConversion

@PostMapping("/cart/add")
@PrismTrackConversion(
    experimentKey = "buy_button_color",
    eventName = "ADD_TO_CART"
)
fun addToCart(
    @PrismUserId @RequestParam userId: String,
    @RequestParam productId: Long
): CartResponse {
    // 장바구니 담기 로직...
    return CartResponse(success = true)
}

    동작 방식: - 메서드가 성공적으로 종료되면 자동으로 전환 이벤트 전송 - 예외 발생 시에는 추적하지 않음 (trackOnException = false 기본값)

    E-commerce 퍼널 예제 (실전)¶

    Sprint 1에서 구현된 실제 E-commerce 퍼널 예제입니다.

    시나리오: 구매 전환율 실험¶

    상품 조회 (실험 참가)
    ↓
장바구니 담기 (ADD_TO_CART 전환)
    ↓
결제 완료 (PURCHASE 전환)

    전체 구현 코드¶

    @RestController
@RequestMapping("/examples/shop")
class ShopFunnelController(
    private val shopCartService: ShopCartService,
    private val conversionTracker: PrismConversionTracker
) {

    // 1. 상품 조회 (실험 참가)
    @GetMapping("/products/{id}")
    @PrismExperiment(experimentKey = "buy_button_color")
    fun getProduct(
        @PathVariable id: Long,
        @PrismUserId @RequestParam userId: String
    ): ShopProductResponse {
        val product = shopService.getProduct(id)
        val variant = PrismContext.getCurrentVariant() ?: "CONTROL"

        return ShopProductResponse.from(product, variant)
    }

    // 2. 장바구니 담기 (ADD_TO_CART 전환)
    @PostMapping("/cart/add")
    @PrismExperiment(experimentKey = "buy_button_color")
    fun addToCart(
        @PrismUserId @RequestParam userId: String,
        @RequestParam productId: Long,
        @RequestParam(defaultValue = "1") quantity: Int
    ): ShopAddToCartResponse {
        val itemCount = shopCartService.addItem(userId, productId, quantity)

        // 안전한 전환 추적 (통계 오염 방지)
        conversionTracker.trackConversionSafe(
            userId = userId,
            experimentKey = "buy_button_color",
            eventName = "ADD_TO_CART"
        )

        return ShopAddToCartResponse(
            message = "Added to cart",
            userId = userId,
            itemCount = itemCount
        )
    }

    // 3. 결제 (PURCHASE 전환)
    @PostMapping("/checkout")
    @PrismExperiment(experimentKey = "buy_button_color")
    fun checkout(
        @PrismUserId @RequestParam userId: String,
        @RequestBody request: ShopCheckoutRequest
    ): ShopCheckoutResponse {
        val total = shopCartService.calculateTotal(request.cartItems)

        conversionTracker.trackConversionSafe(
            userId = userId,
            experimentKey = "buy_button_color",
            eventName = "PURCHASE"
        )

        shopCartService.clearCart(userId)

        return ShopCheckoutResponse(
            message = "Purchase completed",
            userId = userId,
            value = total
        )
    }

    // 4. 어노테이션 기반 전환 추적 (대안)
    @PostMapping("/order/complete")
    @PrismTrackConversion(
        experimentKey = "buy_button_color",
        eventName = "ORDER_COMPLETE"
    )
    fun completeOrder(
        @PrismUserId @RequestParam userId: String,
        @RequestParam orderId: String
    ): String {
        return "Order $orderId completed for user $userId"
    }
}

    테스트 방법¶

    # 1. 상품 조회 (실험 참가)
curl "http://localhost:8090/examples/shop/products/1?userId=user123"

# 2. 장바구니 담기
curl -X POST "http://localhost:8090/examples/shop/cart/add?userId=user123&productId=1&quantity=2"

# 3. 결제
curl -X POST "http://localhost:8090/examples/shop/checkout?userId=user123" \
  -H "Content-Type: application/json" \
  -d '{"cartItems": [{"productId": 1, "quantity": 2}]}'

# 4. 주문 완료 (어노테이션 기반)
curl -X POST "http://localhost:8090/examples/shop/order/complete?userId=user123&orderId=ORD-001"

    전환 추적: 어노테이션 vs 서비스¶

    두 가지 방법으로 전환을 추적할 수 있습니다:

    방법 1: @PrismTrackConversion 어노테이션¶

    @PostMapping("/checkout")
@PrismTrackConversion(
    experimentKey = "buy_button_color",
    eventName = "PURCHASE"
)
fun checkout(@PrismUserId userId: String): Response {
    // 메서드 성공 시 자동 추적
}

    장점: 코드가 간결함, 선언적 단점: 조건부 추적이 어려움

    방법 2: PrismConversionTracker 서비스¶

    @Service
class CheckoutService(
    private val conversionTracker: PrismConversionTracker
) {
    @PrismExperiment(experimentKey = "buy_button_color")
    fun checkout(userId: String, amount: Int) {
        // 비즈니스 로직...

        if (amount > 0) {  // 조건부 추적 가능
            conversionTracker.trackConversionSafe(
                userId = userId,
                experimentKey = "buy_button_color",
                eventName = "PURCHASE"
            )
        }
    }
}

    장점: 조건부 추적 가능, 통계 오염 자동 방지 단점: 코드가 약간 더 길어짐

    주의사항¶

    통계 오염 방지¶

    절대 금지: PrismClient.trackConversion() 직접 호출

    // ❌ 금지! 통계 오염 위험
prismClient.trackConversion(userId, experimentKey, eventName)

// ✅ 권장: 안전한 래퍼 사용
conversionTracker.trackConversionSafe(userId, experimentKey, eventName)

// ✅ 권장: 어노테이션 사용
@PrismTrackConversion(experimentKey = "...", eventName = "...")

    이유: API 장애 시 impression 없이 conversion만 기록되면 통계가 왜곡됩니다.
